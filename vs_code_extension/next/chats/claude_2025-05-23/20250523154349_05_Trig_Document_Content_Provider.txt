import * as vscode from 'vscode';
import { Parser } from 'n3';

export class TrigDocumentProvider implements vscode.TextDocumentContentProvider {
    private _onDidChange = new vscode.EventEmitter<vscode.Uri>();
    readonly onDidChange = this._onDidChange.event;

    constructor(private context: vscode.ExtensionContext) {}

    async provideTextDocumentContent(uri: vscode.Uri): Promise<string> {
        // Extract the actual file URI from the custom scheme
        const actualUri = vscode.Uri.parse(uri.path);
        
        try {
            const document = await vscode.workspace.openTextDocument(actualUri);
            const content = document.getText();
            
            // Add global prefixes for validation
            const globalPrefixes = vscode.workspace.getConfiguration('trigDatasets').get<string>('globalPrefixes', '');
            const fullContent = globalPrefixes + '\n' + content;
            
            // Validate the TriG content
            const isValid = await this.validateTriGContent(fullContent);
            
            if (isValid) {
                return this.generateValidationSuccessPage(actualUri);
            } else {
                return this.generateValidationFailurePage(actualUri);
            }
        } catch (error) {
            return this.generateValidationFailurePage(actualUri, error as Error);
        }
    }

    private async validateTriGContent(content: string): Promise<boolean> {
        return new Promise((resolve) => {
            const parser = new Parser({ format: 'TriG' });
            let hasError = false;
            
            parser.parse(content, (error, quad, prefixes) => {
                if (error) {
                    hasError = true;
                    resolve(false);
                } else if (!quad && !hasError) {
                    // Parsing complete without errors
                    resolve(true);
                }
            });
        });
    }
